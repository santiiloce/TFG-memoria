Predicción conforme. La predicción conforme (CP) se erige como un marco estadístico robusto en el aprendizaje automático, diseñado principalmente para proporcionar medidas de confianza para las predicciones (Shafer y Vovk~\cite{shafer2008}; Angelopoulos y Bates~\cite{angelopoulos2021}).

\pendiente{definición}
\end{definicion}

\begin{definicion}[Predicción conforme dividida]\label{def:split-conformal}
% Puntuación de no conformidad s(x, y); puntuaciones de calibración s_1, ..., s_n;
% \hat q = k-ésimo menor valor, k = \lceil (n+1)(1-\alpha) \rceil (\hat q = \infty si k > n);
% C(x) = \{ y : s(x, y) \le \hat q \}.
\pendiente{definición}
\end{definicion}

\begin{teorema}[Cobertura marginal]\label{teo:cobertura}
% Si (X_1, Y_1), ..., (X_{n+1}, Y_{n+1}) son intercambiables y \hat q es el k-ésimo menor valor de las
% puntuaciones de calibración, k = \lceil (n+1)(1-\alpha) \rceil, entonces
% 1 - \alpha \le \Prob(Y_{n+1} \in C(X_{n+1})) \le 1 - \alpha + 1/(n+1) (la cota superior, sin empates).
\pendiente{enunciado}
\end{teorema}
